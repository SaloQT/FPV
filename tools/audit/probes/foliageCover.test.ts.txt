// Probe, not a test: copy to src/render/vegetation/zzcover.test.ts, run npx vitest run on it, then delete it. It fails on purpose and prints, per
// variant, near-LOD triangles, pieces, drop share and side-view cover (painted atlas alpha at 1.5 cm cells). 1 - cover is optimistic: the
// sky fraction tools/audit/skyfrac.py reads from a render of the same tree is several points higher.
import { it } from 'vitest';
import type { Piece } from './foliageSpace';
import { KIND } from './meshBuilder';
import { buildTreeLods } from './treeGen';
import { VARIANT_DEFS, variantPlan } from './variants';
import { TILES_X, TILES_Y, buildLeafAtlas } from './leafAtlas';

// Side-view cover of a tree's near LOD: painted foliage (each piece through its own paint mask) and wood, projected along z onto a grid of
// `cell` metres. Cover = covered cells / cells inside each row's silhouette span (leftmost to rightmost covered), over the crown rows.
it('diag cover', () => {
  const out: string[] = [];
  const atlas = buildLeafAtlas(512), AW = 512 * TILES_X, AH = 512 * TILES_Y, alpha = atlas.colour[0];
  const painted = (u: number, v: number): boolean => alpha[(Math.min(AH - 1, Math.max(0, Math.floor(v * AH))) * AW + Math.min(AW - 1, Math.max(0, Math.floor(u * AW)))) * 4 + 3] >= 128;
  for (const name of ['spruce open', 'pine open']) {
    const v = VARIANT_DEFS.findIndex((d) => d.name === name);
    let all: Piece[] = [], drop = 0;
    const mesh = buildTreeLods(variantPlan(v)!, VARIANT_DEFS[v].seed, undefined, { pieces: (p) => { all = p; }, report: (s) => { drop = s.dropped / (s.placed + s.dropped); } })[0];
    const cell = 0.015, cov = new Map<number, Set<number>>();
    const mark = (x: number, y: number): void => { const r = Math.floor(y / cell); let s = cov.get(r); if (!s) cov.set(r, (s = new Set())); s.add(Math.floor(x / cell)); };
    let foliageArea = 0;
    for (const p of all) for (const [a, b, c] of p.tris) {
      const A = p.pos[a], B = p.pos[b], C = p.pos[c];
      const L = Math.max(Math.hypot(B[0] - A[0], B[1] - A[1], B[2] - A[2]), Math.hypot(C[0] - A[0], C[1] - A[1], C[2] - A[2]));
      const n = Math.max(2, Math.ceil(L / (cell / 3)));
      for (let i = 0; i <= n; i++) for (let j = 0; j <= n - i; j++) {
        const s = i / n, t = j / n, w = 1 - s - t;
        const u = p.uv[a][0] * w + p.uv[b][0] * s + p.uv[c][0] * t, vv = p.uv[a][1] * w + p.uv[b][1] * s + p.uv[c][1] * t;
        if (!painted(u, vv)) continue;
        mark(A[0] * w + B[0] * s + C[0] * t, A[1] * w + B[1] * s + C[1] * t);
      }
      foliageArea += 0.5 * Math.hypot(...([0, 1, 2].map((k) => { const e1 = [B[0] - A[0], B[1] - A[1], B[2] - A[2]], e2 = [C[0] - A[0], C[1] - A[1], C[2] - A[2]]; return [e1[1] * e2[2] - e1[2] * e2[1], e1[2] * e2[0] - e1[0] * e2[2], e1[0] * e2[1] - e1[1] * e2[0]][k]; })) as [number, number, number]);
    }
    for (let t = 0; t < mesh.idx.length; t += 3) {
      const ids = [mesh.idx[t], mesh.idx[t + 1], mesh.idx[t + 2]];
      if (mesh.attr[ids[0] * 4 + 2] >= KIND.leaf) continue;
      const P = ids.map((i) => [mesh.pos[i * 3], mesh.pos[i * 3 + 1]]);
      const L = Math.max(Math.hypot(P[1][0] - P[0][0], P[1][1] - P[0][1]), Math.hypot(P[2][0] - P[0][0], P[2][1] - P[0][1]));
      const n = Math.max(2, Math.ceil(L / (cell / 2)));
      for (let i = 0; i <= n; i++) for (let j = 0; j <= n - i; j++) { const s = i / n, q = j / n, w = 1 - s - q; mark(P[0][0] * w + P[1][0] * s + P[2][0] * q, P[0][1] * w + P[1][1] * s + P[2][1] * q); }
    }
    const plan = variantPlan(v)!, y0 = plan.crownC[1] - plan.crownR[1], y1 = plan.crownC[1] + plan.crownR[1];
    let inside = 0, covered = 0;
    for (const [r, s] of cov) {
      const y = (r + 0.5) * cell;
      if (y < y0 + 0.1 * (y1 - y0) || y > y1 - 0.1 * (y1 - y0)) continue;
      let lo = Infinity, hi = -Infinity; for (const x of s) { lo = Math.min(lo, x); hi = Math.max(hi, x); }
      inside += hi - lo + 1; covered += s.size;
    }
    out.push(`${name} tris ${mesh.idx.length / 3} pieces ${all.length} drop ${drop.toFixed(3)} foliage m2 ${foliageArea.toFixed(0)} cover ${(covered / inside).toFixed(3)}`);
  }
  throw new Error(out.join(' || '));
}, 600000);
